cmake_minimum_required(VERSION 3.16)
# Optional offline build using an existing, version-matched IDF component cache.
if(DEFINED ENV{DEMO_COMPONENT_CACHE})
    set(ENV{IDF_COMPONENT_MANAGER} 0)
    foreach(component esp32-camera esp_jpeg led_strip esp_codec_dev)
        list(APPEND EXTRA_COMPONENT_DIRS "$ENV{DEMO_COMPONENT_CACHE}/espressif__${component}")
    endforeach()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(esp32s3_camera_display_demo)
if(DEFINED ENV{DEMO_COMPONENT_CACHE})
    idf_component_get_property(camera_target espressif__esp32-camera COMPONENT_LIB)
    target_link_libraries(${camera_target} PUBLIC idf::espressif__esp_jpeg)
endif()
